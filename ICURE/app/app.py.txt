@st.cache_resource
def load_model():
    return joblib.load(MODEL_PATH)


def to_label(c):
    s = str(c)
    if s in ("N", "S", "V"):
        return s
    return {"0": "N", "1": "S", "2": "V"}.get(s, s)


def predict_rr(model, rr_values):
    rr = np.asarray(rr_values, dtype=float)
    if hasattr(model, "feature_names_in_"):
        X = pd.DataFrame({model.feature_names_in_[0]: rr})
    else:
        X = rr.reshape(-1, 1)

    preds = [to_label(p) for p in model.predict(X)]
    probs = model.predict_proba(X)
    labels = [to_label(c) for c in model.classes_]
    return preds, probs, labels


def send_caregiver_alert(prediction, rr_seconds, probabilities, labels):
    secrets_file = BASE_DIR / ".streamlit" / "secrets.toml"

    if not secrets_file.exists():
        return False, "Telegram settings file not found."

    try:
        with open(secrets_file, "rb") as file:
            settings = tomllib.load(file)

        token = settings["TELEGRAM_BOT_TOKEN"]
        chat_id = settings["TELEGRAM_CHAT_ID"]

    except (OSError, KeyError, tomllib.TOMLDecodeError):
        return False, "Telegram settings are incomplete or invalid."

    probability_text = "\n".join(
        f"{label}: {probability * 100:.1f}%"
        for label, probability in zip(labels, probabilities)
    )

    class_description = CLASS_NAMES.get(prediction, prediction)

    message = (
        "CARDIOSENSE AI — EXPERIMENTAL SCREENING NOTIFICATION\n\n"
        f"The prototype model flagged: {prediction} ({class_description})\n"
        f"Previous RR interval: {rr_seconds:.2f} seconds\n"
        f"Approximate heart-rate equivalent: {60 / rr_seconds:.0f} bpm\n\n"
        f"Model probability output:\n{probability_text}\n\n"
        "This is an experimental software prediction, NOT a diagnosis or confirmed emergency. "
        "Please check on the person and follow appropriate medical guidance if they feel unwell."
    )

    url = f"https://api.telegram.org/bot{token}/sendMessage"
    payload = urllib.parse.urlencode({
        "chat_id": chat_id,
        "text": message
    }).encode("utf-8")

    request = urllib.request.Request(url, data=payload)

    try:
        with urllib.request.urlopen(request, timeout=15) as response:
            result = json.loads(response.read().decode("utf-8"))

        if result.get("ok"):
            return True, "Telegram notification delivered to the caregiver."

        return False, "Telegram did not accept the notification."

    except Exception as error:
        return False, f"Telegram delivery failed ({type(error).__name__})."


def maybe_send_caregiver_alert(prediction, rr_seconds, probabilities, labels):
    if prediction == "N":
        return None

    now = time.time()
    last_sent = st.session_state.get("caregiver_alert_last_sent", {})
    previous = last_sent.get(prediction, 0)
    cooldown_seconds = 300

    if now - previous < cooldown_seconds:
        remaining = int(cooldown_seconds - (now - previous))
        return (
            "cooldown",
            f"A {prediction}-class notification was already sent recently. "
            f"Cooldown: {remaining // 60}m {remaining % 60}s remaining."
        )

    success, message = send_caregiver_alert(
        prediction, rr_seconds, probabilities, labels
    )

    if success:
        last_sent[prediction] = now
        st.session_state["caregiver_alert_last_sent"] = last_sent
        return ("success", message)

    return ("error", message)

def style_fig(fig, height=340):
    fig.update_layout(
        paper_bgcolor="rgba(0,0,0,0)",
        plot_bgcolor="rgba(0,0,0,0)",
        font=dict(family="JetBrains Mono, monospace", color="#E5E7EB"),
        margin=dict(l=10, r=10, t=40, b=10),
        height=height,
        legend=dict(bgcolor="rgba(0,0,0,0)"),
    )
    fig.update_xaxes(
        gridcolor="rgba(139,92,246,0.15)",
        zerolinecolor="rgba(139,92,246,0.25)"
    )
    fig.update_yaxes(
        gridcolor="rgba(139,92,246,0.15)",
        zerolinecolor="rgba(139,92,246,0.25)"
    )
    return fig


def distribution_chart(counts, title, log=False):
    letters = ["N", "S", "V"]
    fig = go.Figure(
        go.Bar(
            x=letters,
            y=[counts.get(k, 0) for k in letters],
            marker=dict(color=[CLASS_COLORS[k] for k in letters]),
            text=[f"{counts.get(k, 0):,}" for k in letters],
            textposition="outside",
        )
    )
    fig.update_layout(title=title)
    if log:
        fig.update_yaxes(type="log")
    return style_fig(fig)


def prob_chart(probs, labels):
    fig = go.Figure(
        go.Bar(
            y=labels,
            x=[p * 100 for p in probs],
            orientation="h",
            marker=dict(color=[CLASS_COLORS.get(l, PURPLE) for l in labels]),
            text=[f"{p*100:.1f}%" for p in probs],
            textposition="outside",
        )
    )
    fig.update_layout(title="Model probability by class")
    fig.update_xaxes(range=[0, 115], title="%")
    return style_fig(fig, height=260)


def illustrative_ecg():
    t = np.linspace(0, 3, 1500)
    y = np.zeros_like(t)

    for c in (0.4, 1.4, 2.4):
        y += 0.12 * np.exp(-((t - (c - 0.18)) ** 2) / 0.0025)
        y -= 0.15 * np.exp(-((t - (c - 0.04)) ** 2) / 0.00015)
        y += 1.0 * np.exp(-((t - c) ** 2) / 0.00012)
        y -= 0.22 * np.exp(-((t - (c + 0.04)) ** 2) / 0.00015)
        y += 0.28 * np.exp(-((t - (c + 0.28)) ** 2) / 0.004)

    fig = go.Figure(
        go.Scatter(x=t, y=y, mode="lines", line=dict(color=LIME, width=2.5))
    )
    fig.update_layout(title="Illustrative ECG waveform (not a real recording)")
    fig.update_xaxes(title="time (s)")
    fig.update_yaxes(title="amplitude (a.u.)")
    return style_fig(fig, height=300)
