import json
import urllib.request
from getpass import getpass

print("CARDIOSENSE AI - TELEGRAM SETUP")

token = getpass("Enter your Telegram bot API token: ").strip()

url = f"https://api.telegram.org/bot{token}/getUpdates"

try:
    request = urllib.request.Request(url)

    with urllib.request.urlopen(request, timeout=15) as response:
        data = json.loads(response.read().decode())

    if not data.get("ok"):
        print("Telegram API returned an error.")

    elif not data.get("result"):
        print("No messages found.")
        print("Open your bot and send /start again.")

    else:
        found = False

        for update in data["result"]:
            message = update.get("message", {})
            chat = message.get("chat", {})

            if chat and chat.get("type") == "private":
                print("\nCaregiver detected!")
                print("Name:", chat.get("first_name", "Unknown"))
                print("Chat ID:", chat.get("id"))
                found = True

        if not found:
            print("No private chat found. Send /start again.")

except Exception:
    print("Connection failed. Check your internet and bot token.")

finally:
    token = None
